package models;

import interfaces.Comprable;

public abstract class Producto implements Comprable {
    private String nombre;
    private double precio;

    public Producto(String nombre, double precio) {
        this.nombre = nombre;
        this.precio = precio;
    }

    public String getNombre() { return nombre; }
    public double getPrecio() { return precio; }

    // Método abstracto que cada producto implementará
    public abstract String getDescripcion();

package models;

public class Videojuego extends Producto {
    private String genero;

    public Videojuego(String nombre, String genero, double precio) {
        super(nombre, precio);
        this.genero = genero;
    }

    @Override
    public String getDescripcion() {
        return "Videojuego: " + getNombre() + " | Género: " + genero + " | Precio: $" + getPrecio();
    }
}

Clase Biblioteca.java (con arreglo)

package services;

import models.Videojuego;

public class Biblioteca {
    private Videojuego[] juegos;
    private int contador;

    public Biblioteca(int capacidad) {
        juegos = new Videojuego[capacidad];
        contador = 0;
    }

    public void agregarJuego(Videojuego juego) {
        if (contador < juegos.length) {
            juegos[contador] = juego;
            contador++;
            System.out.println("✅ Juego agregado: " + juego.getNombre());
        } else {
            System.out.println("⚠️ Biblioteca llena. No se puede agregar más.");
        }
    }

    public void mostrarJuegos() {
        System.out.println("\n📚 Juegos en la Biblioteca:");
        for (int i = 0; i < contador; i++) {
            System.out.println(" - " + juegos[i].getDescripcion());
        }
    }
}

package app;
